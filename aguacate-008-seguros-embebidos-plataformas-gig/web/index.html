// integration

Para tu equipo de ingeniería.

01

Instala el SDK

npm install amparo-sdk

import { Amparo } from 'amparo-sdk';
const client = new Amparo({ apiKey: process.env.AMPARO_KEY });
02

Activa cobertura por turno

const coverage = await client.shifts.start({
  contractor_id: driver.id,
  shift_duration_min: 480,
  coverage_tier: 'accidente_personal+vida',
});
03

Recibe webhooks de claims

app.post('/webhooks/amparo', (req, res) => {
  const { event, claim } = req.body;
  if (event === 'claim.approved') notifyContractor(claim);
});
